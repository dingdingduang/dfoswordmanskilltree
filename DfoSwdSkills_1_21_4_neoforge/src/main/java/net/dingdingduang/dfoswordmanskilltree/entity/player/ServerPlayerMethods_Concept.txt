    public static Map<ServerPlayer, Integer> Player_IDMap = new TreeMap<>();

    public static ServerPlayer GetPlayerByID(int a) {
        plUpdate();
        return ID_PlayerMap.get(a);
    }

    public static int GetPlayerIDbyServerPlayer(ServerPlayer a) {
        plUpdate();
        return Player_IDMap.get(a);
    }

    public static void plUpdate() {
        PlayerList playerlist1 = getMinecraftServerInstance().getPlayerList();

        if (!pl1.equals(playerlist1)) {
            ID_PlayerMap = new TreeMap<>();
            Player_IDMap = new TreeMap<>();
            List<ServerPlayer> serverplayerlist1 = playerlist1.getPlayers();
            int tempi = 0;
            Iterator<ServerPlayer> var = serverplayerlist1.iterator();

            ServerPlayer serverplayer;
            while (var.hasNext()) {
                serverplayer = var.next();
                ID_PlayerMap.put(tempi, serverplayer);
                Player_IDMap.put(serverplayer, tempi);
            }
            //while (!serverplayer.getGameProfile().getName().equalsIgnoreCase(pUsername));
        }
    }
