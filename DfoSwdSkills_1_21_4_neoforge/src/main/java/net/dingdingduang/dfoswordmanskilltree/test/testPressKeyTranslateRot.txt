    @SubscribeEvent
    public static void registerKeyMappingsEvent2(RegisterKeyMappingsEvent event) {
        event.register(STBasicLibs_TestTranslateEventXLeft);
        event.register(STBasicLibs_TestTranslateEventXRight);
        event.register(STBasicLibs_TestTranslateEventZForward);
        event.register(STBasicLibs_TestTranslateEventZBackward);
        event.register(STBasicLibs_TestTranslateEventYUp);
        event.register(STBasicLibs_TestTranslateEventYDown);

        event.register(STBasicLibs_TestTranslateEventXROTUP);
        event.register(STBasicLibs_TestTranslateEventXROTUPDOWN);
        event.register(STBasicLibs_TestTranslateEventYROTUP);
        event.register(STBasicLibs_TestTranslateEventYROTUPDOWN);
        event.register(STBasicLibs_TestTranslateEventZROTUP);
        event.register(STBasicLibs_TestTranslateEventZROTUPDOWN);
    }

    public static float tempFloatNewX = 0;
    public static float tempFloatNewY = 0;
    public static float tempFloatNewZ = 0;

    public static float tempFloatNewRotX = 0;
    public static float tempFloatNewRotY = 0;
    public static float tempFloatNewRotZ = 0;

    @EventBusSubscriber(modid = DfoSwordmanSkillTreeConstants.MOD_ID, value = Dist.CLIENT)
    private static class InputEvents2 {
        @SubscribeEvent
        public static void inputEvent2(InputEvent.Key event) {
            Minecraft minecraft = getMinecraftInstance();
            Player player = minecraft.player;
            if (player == null || minecraft.screen != null) return;
//            if (event.getKey() == STBasicLibs_TestKeyEvent.getKey().getValue()) {
            if (STBasicLibs_TestTranslateEventXLeft.isDown()) {
                tempFloatNewX = tempFloatNewX - 0.5f;
                printInGameMsg("newX: "+tempFloatNewX);
            }
            if (STBasicLibs_TestTranslateEventXRight.isDown()) {
                tempFloatNewX = tempFloatNewX + 0.5f;
                printInGameMsg("newX: "+tempFloatNewX);
            }
            if (STBasicLibs_TestTranslateEventZForward.isDown()) {
                tempFloatNewY = tempFloatNewY - 0.5f;
                printInGameMsg("newY: "+tempFloatNewY);
            }
            if (STBasicLibs_TestTranslateEventZBackward.isDown()) {
                tempFloatNewY = tempFloatNewY + 0.5f;
                printInGameMsg("newY: "+tempFloatNewY);
            }
            if (STBasicLibs_TestTranslateEventYDown.isDown()) {
                tempFloatNewZ = tempFloatNewZ - 0.5f;
                printInGameMsg("newZ: "+tempFloatNewZ);
            }
            if (STBasicLibs_TestTranslateEventYUp.isDown()) {
                tempFloatNewZ = tempFloatNewZ + 0.5f;
                printInGameMsg("newZ: "+tempFloatNewZ);
            }



            if (STBasicLibs_TestTranslateEventXROTUP.isDown()) {
                tempFloatNewRotX = tempFloatNewRotX + 15f;
                printInGameMsg("new XROT: "+tempFloatNewRotX);
            }
            if (STBasicLibs_TestTranslateEventXROTUPDOWN.isDown()) {
                tempFloatNewRotX = tempFloatNewRotX - 15f;
                printInGameMsg("new XROT: "+tempFloatNewRotX);
            }
            if (STBasicLibs_TestTranslateEventYROTUP.isDown()) {
                tempFloatNewRotY = tempFloatNewRotY + 15f;
                printInGameMsg("new YROT: "+tempFloatNewRotY);
            }
            if (STBasicLibs_TestTranslateEventYROTUPDOWN.isDown()) {
                tempFloatNewRotY = tempFloatNewRotY - 15f;
                printInGameMsg("new YROT: "+tempFloatNewRotY);
            }
            if (STBasicLibs_TestTranslateEventZROTUP.isDown()) {
                tempFloatNewRotZ = tempFloatNewRotZ + 15f;
                printInGameMsg("new ZROT: "+tempFloatNewRotZ);
            }
            if (STBasicLibs_TestTranslateEventZROTUPDOWN.isDown()) {
                tempFloatNewRotZ = tempFloatNewRotZ - 15f;
                printInGameMsg("new ZROT: "+tempFloatNewRotZ);
            }
        }

        //releasing a pressed key
        private static boolean isKeyReleased(InputEvent.Key e) {
            return e.getAction() == 0;
        }

        //pressing down on a key
        private static boolean isKeyDown(InputEvent.Key e) {
            return e.getAction() == 1;
        }

        //holding a key
        private static boolean isKeyHolding(InputEvent.Key e) {
            return e.getAction() == 2;
        }
    }
}
