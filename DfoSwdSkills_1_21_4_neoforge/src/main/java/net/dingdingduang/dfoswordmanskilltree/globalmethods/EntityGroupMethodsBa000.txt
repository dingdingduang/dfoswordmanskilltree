    public static void MoveAllEntitiesInEntityGroup(List<Entity> a, Vector3f b, boolean resetDelta) {
        for (Entity tempEntity: a) {
            if (tempEntity == null) { continue; }
            if (tempEntity instanceof LivingEntity tempLivingEntity) {

                moveEntityToPos(tempLivingEntity, b);
                if (resetDelta) {
                    setEntityDeltaMovement(tempLivingEntity, 0, 0, 0);
                }
            }
        }
    }

    public static void MoveAllEntitiesInEntityGroupHorizon(List<Entity> a, float dist, float horizontalDeg, boolean resetDelta) {
        for (Entity tempEntity: a) {
            if (tempEntity == null) { continue; }
            if (tempEntity instanceof LivingEntity tempLivingEntity) {

                moveEntityToPos(tempLivingEntity, PolarProjectionFromVecHorizon(getEntityPos(tempLivingEntity), dist, horizontalDeg));
                if (resetDelta) {
                    setEntityDeltaMovement(tempLivingEntity, 0, 0, 0);
                }
            }
        }
    }

    //basic
    public static void DamageEntityGroup(List<Entity> a, LivingEntity dealer, float amount, boolean isMagic) {
        int iframeTime;
        for (Entity tempEntity: a) {
            if (tempEntity == null) { continue; }
            if (tempEntity instanceof LivingEntity tempLivingEntity) {

                addKnockbackResist(tempLivingEntity);
                iframeTime = GetEntityInvulnerableTime(tempLivingEntity);
                SetEntityInvulnerableTime(tempLivingEntity, 0);

                if (isMagic) {
                    LivingEntityMagicDamageTarget(dealer, tempLivingEntity, amount);
                }
                else {
                    LivingEntityDamageTarget(dealer, tempLivingEntity, amount);
                }

                removeKnockbackResist(tempLivingEntity);
                SetEntityInvulnerableTime(tempLivingEntity, iframeTime);
            }
        }
    }

    //Advance
    public static void DamageEntityGroup(List<Entity> a, LivingEntity dealer, float amount, String FXNamespace, String FXFilename, double FXHeight, boolean BindToDMGEntity, boolean isMagic) {
        int iframeTime;
        for (Entity tempEntity: a) {
            if (tempEntity == null) { continue; }
            if (tempEntity instanceof LivingEntity tempLivingEntity) {
                addKnockbackResist(tempLivingEntity);
                iframeTime = GetEntityInvulnerableTime(tempLivingEntity);
                SetEntityInvulnerableTime(tempLivingEntity, 0);

                if (isMagic) {
                    LivingEntityMagicDamageTarget(dealer, tempLivingEntity, amount);
                }
                else {
                    LivingEntityDamageTarget(dealer, tempLivingEntity, amount);
                }

                removeKnockbackResist(tempLivingEntity);
                SetEntityInvulnerableTime(tempLivingEntity, iframeTime);

                if (FXNamespace != null && FXFilename != null) {
                    if (BindToDMGEntity) {
                        BindEffectToEntity(tempLivingEntity, FXNamespace, FXFilename);
                    }
                    else {
                        ProjectileDummyEffOnly tempEffEntity = SetupEffOnlyEntity(getEntityPos(tempLivingEntity), getEntityHorizontalFacingDeg(tempLivingEntity));
                        tempEffEntity.increaseEntityHeightBy(FXHeight);
                        SpawnEntity(tempEffEntity);
                        BindEffectToEntity(tempEffEntity, FXNamespace, FXFilename);
                    }
                }
            }
        }
    }

        int iframeTime;
        for (Entity tempEntity: a) {
            if (tempEntity == null) {
                continue;
            }
            if (tempEntity instanceof LivingEntity tempLivingEntity) {

                entityHitCounter.put(tempLivingEntity, entityHitCounter.containsKey(tempLivingEntity) ? entityHitCounter.get(tempLivingEntity)+1 : 1);

                addKnockbackResist(tempLivingEntity);
                iframeTime = GetEntityInvulnerableTime(tempLivingEntity);
                SetEntityInvulnerableTime(tempLivingEntity, 0);

                if (!effectMap.isEmpty()) {
                    for (MobEffectInstance j : effectMap.keySet()) {
                        applyEffectToLivingEntity(tempLivingEntity, j);
                    }
                }

                if (isMagic) {
                    LivingEntityMagicDamageTarget(dealer, tempLivingEntity, amount);
                } else {
                    LivingEntityDamageTarget(dealer, tempLivingEntity, amount);
                }

                LivingEntityDamageTarget(dealer, tempLivingEntity, amount);

                removeKnockbackResist(tempLivingEntity);
                SetEntityInvulnerableTime(tempLivingEntity, iframeTime);
            }
        }
    }
