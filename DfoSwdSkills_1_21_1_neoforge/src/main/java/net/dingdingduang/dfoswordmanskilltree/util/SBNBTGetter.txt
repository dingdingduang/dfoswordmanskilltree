    //U cannot modify player data
    public static void setLivingEntityMaxHP(Entity a, int b) {
        CompoundTag ct = getEntityCompoundNBT(a);
        CompoundTag nbtEntityTag2 = new CompoundTag();
        nbtEntityTag2.putDouble("Base", b);
        nbtEntityTag2.putString("Name","minecraft:generic.max_health");
        ListTag LT1 = new ListTag();
        LT1.add(nbtEntityTag2);
        ct.put("Attributes", LT1);
        ct.putInt("Health",b);
        try {
            getEntityDataAccessor(a).setData(ct);
        } catch (CommandSyntaxException e) {
            throw new RuntimeException(e);
        }
    }

    public static CompoundTag getEntityCompoundNBT(Entity a) {
        return getEntityDataAccessor(a).getData();
    }

    public static EntityDataAccessor getEntityDataAccessor(Entity a) {
        return new EntityDataAccessor(a);
    }
}
