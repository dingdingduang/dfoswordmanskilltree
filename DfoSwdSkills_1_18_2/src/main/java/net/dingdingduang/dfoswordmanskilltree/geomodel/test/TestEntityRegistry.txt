package net.dingdingduang.dfoswordmanskilltree.geomodel.test;

import net.dingdingduang.dfoswordmanskilltree.DfoSwordmanSkillTreeConstants;

import net.dingdingduang.dfoswordmanskilltree.geomodel.test.car.TestCarEntity;
import net.minecraft.resources.ResourceLocation;
import net.minecraft.world.entity.EntityType;
import net.minecraft.world.entity.MobCategory;
import net.minecraftforge.eventbus.api.IEventBus;
import net.minecraftforge.registries.DeferredRegister;
import net.minecraftforge.registries.ForgeRegistries;
import net.minecraftforge.registries.RegistryObject;

public class TestEntityRegistry {
    public static final DeferredRegister<EntityType<?>> ENTITIES = DeferredRegister.create(ForgeRegistries.ENTITIES,
            DfoSwordmanSkillTreeConstants.MOD_ID);

    public static final RegistryObject<EntityType<TestBikeEntity>> TEST_BIKE_ENTITY =
            ENTITIES.register("testbikeabcdefg",
                    () -> EntityType.Builder.<TestBikeEntity>of(TestBikeEntity::new, MobCategory.MISC)
                            .sized(1.0f, 0f)
                            .setUpdateInterval(Integer.MAX_VALUE)
                            .build(new ResourceLocation(DfoSwordmanSkillTreeConstants.MOD_ID, "testbikeabcdefg").toString()));

    public static final RegistryObject<EntityType<TestCarEntity>> TEST_CAR_ENTITY =
            ENTITIES.register("testcarrrabcdefg",
                    () -> EntityType.Builder.<TestCarEntity>of(TestCarEntity::new, MobCategory.CREATURE)
                            .sized(0.5f, 0.6f)
                            .build(new ResourceLocation(DfoSwordmanSkillTreeConstants.MOD_ID, "testcarrrabcdefg").toString()));

    public static void register(IEventBus eventBus) {
        ENTITIES.register(eventBus);
    }
}
