package net.dingdingduang.dfoswordmanskilltree.networking.packet.toclientpacket;

import net.minecraft.network.FriendlyByteBuf;
import net.minecraftforge.network.NetworkEvent;

import java.util.function.Supplier;

import static net.dingdingduang.dfoswordmanskilltree.globalmethods.DfoGeneralMethods.getClientPlayer;

public class FetchClientPlayerHasGravityFromServer {
    private final boolean hasGravity;

    public FetchClientPlayerHasGravityFromServer(boolean hasGravity) {
        this.hasGravity = hasGravity;
    }

    public FetchClientPlayerHasGravityFromServer(FriendlyByteBuf buf) {
        this.hasGravity = buf.readBoolean();
    }

    public void toBytes(FriendlyByteBuf buf) {
        buf.writeBoolean(this.hasGravity);
    }

    public boolean handle(Supplier<NetworkEvent.Context> ctx) {
        ctx.get().enqueueWork(() -> {
            //do shiet on client side
            if (getClientPlayer() == null) { return; }
            getClientPlayer().setNoGravity(!this.hasGravity);
        });

        return true;
    }
}
