package net.dingdingduang.dfoswordmanskilltree.particle;

import net.minecraft.client.multiplayer.ClientLevel;
import net.minecraft.client.particle.*;
import net.minecraft.core.particles.SimpleParticleType;
import net.minecraftforge.api.distmarker.Dist;
import net.minecraftforge.api.distmarker.OnlyIn;

public class ParticleExtension extends BaseAshSmokeParticle {
    protected ParticleExtension(ClientLevel pLevel, double pX, double pY, double pZ, double pXSpeed, double pYSpeed, double pZSpeed, float pQuadSizeMultiplier, SpriteSet pSprites) {
        super(pLevel, pX, pY, pZ, 0F, 0F, 0F, pXSpeed, pYSpeed, pZSpeed, pQuadSizeMultiplier, pSprites, 0.5F, 20, 0F, false);
    }

//    @Override
//    public void render(VertexConsumer pBuffer, Camera pRenderInfo, float pPartialTicks) {
//
//    }

    public void tick() {
        super.tick();
    }

    @Override
    public ParticleRenderType getRenderType() {
        return ParticleRenderType.PARTICLE_SHEET_TRANSLUCENT;
    }

    @OnlyIn(Dist.CLIENT)
    public static class ParticleExtensionSprite implements ParticleProvider<SimpleParticleType> {
        private final SpriteSet sprites;

        public ParticleExtensionSprite(SpriteSet pSprites) {
            this.sprites = pSprites;
        }

        @Override
        public Particle createParticle(SimpleParticleType pType, ClientLevel pLevel, double pX, double pY, double pZ, double pXSpeed, double pYSpeed, double pZSpeed) {
            return new ParticleExtension(pLevel, pX, pY, pZ, 0.0, 0.0, 0.0, 1.0F, this.sprites);
        }
    }
}
