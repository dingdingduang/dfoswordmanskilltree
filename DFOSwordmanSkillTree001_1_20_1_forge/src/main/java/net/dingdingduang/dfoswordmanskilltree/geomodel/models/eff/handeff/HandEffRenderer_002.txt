    private void renderHandPrep(HandEffEntity handEffEntity, LivingEntity pLivingEntity, PoseStack pPoseStack, float pPartialTicks, float entityYaw, MultiBufferSource bufferSource, int packedLight) {
        pPoseStack.pushPose();
        Matrix4f HandEffRotationMatrix = new Matrix4f();

        if (!SBConditions.isSpectator(pLivingEntity)) {
            HumanoidModel<?> humanoidModel = getOwnerHumanoidModel(pLivingEntity);
            if (humanoidModel != null) {
                float entityBodyYRotLerp = Vector3fMethods.lerp1D(pPartialTicks, pLivingEntity.yBodyRotO, pLivingEntity.yBodyRot);

                if (pLivingEntity.hasPose(Pose.SLEEPING)) {
                    Direction direction = pLivingEntity.getBedOrientation();
                    if (direction != null) {
                        float f4 = pLivingEntity.getEyeHeight(Pose.STANDING) - 0.1F;
                        pPoseStack.translate((float)(-direction.getStepX()) * f4, 0.0F, (float)(-direction.getStepZ()) * f4);
                        HandEffRotationMatrix.translate((float)(-direction.getStepX()) * f4, 0.0F, (float)(-direction.getStepZ()) * f4);
                    }
                }

                setupRotations(pLivingEntity, pPoseStack, entityBodyYRotLerp, pPartialTicks, HandEffRotationMatrix);
                pPoseStack.scale(-1.0F, -1.0F, 1.0F);
                HandEffRotationMatrix.scale(-1.0F, -1.0F, 1.0F);
                scale(pLivingEntity, pPoseStack, pPartialTicks, HandEffRotationMatrix);
                pPoseStack.translate(0.0F, -1.501F, 0.0F);
                HandEffRotationMatrix.translate(0.0F, -1.501F, 0.0F);

                ModelPart bone = humanoidModel.leftArm;
                setuptranslate(pPoseStack, bufferSource, pLivingEntity, handEffEntity, entityYaw, bone, packedLight, pPartialTicks, HandEffRotationMatrix);
            }
        }

        pPoseStack.popPose();
    }

    private static HumanoidModel<?> getOwnerHumanoidModel(LivingEntity owner) {
        EntityRenderer<? super LivingEntity> entityRenderer = getMinecraftInstance().getEntityRenderDispatcher().getRenderer(owner);
        if (entityRenderer instanceof LivingEntityRenderer<?, ?> livingRenderer) {
            EntityModel<?> entityModel = livingRenderer.getModel();

            if (entityModel instanceof HumanoidModel<?> humanoidModel) {
                return humanoidModel;
            }
        }
        return null;
    }

    public void setuptranslate(PoseStack pPoseStack, MultiBufferSource bufferSource, LivingEntity pLivingEntity, HandEffEntity handEffEntity, float pRotationYaw, ModelPart bone, int packedLight, float partialTicks, Matrix4f HandEffRotationMatrix) {
        pPoseStack.pushPose();
        pPoseStack.translate(bone.x * DfoSwordmanSkillTreeConstants.OneOver16, bone.y * DfoSwordmanSkillTreeConstants.OneOver16, bone.z * DfoSwordmanSkillTreeConstants.OneOver16);

//        printInGameMsg("x: "+bone.x+", y: "+bone.y+", z: "+bone.z);

        setupRotation(pPoseStack, bufferSource, pLivingEntity, handEffEntity, pRotationYaw, bone, packedLight, partialTicks, HandEffRotationMatrix);
        pPoseStack.popPose();
    }

    public void setupRotation(PoseStack pPoseStack, MultiBufferSource bufferSource, LivingEntity pLivingEntity, HandEffEntity handEffEntity, float pRotationYaw, ModelPart bone, int packedLight, float partialTicks, Matrix4f HandEffRotationMatrix) {
        pPoseStack.pushPose();

        Quaternionf rotator;
        if (bone.zRot != 0) {
            rotator = new Quaternionf();
            Vector3f AxisZN = new Vector3f(0, 0, 1);
            rotator.rotationAxis(bone.zRot, AxisZN);
//            pPoseStack.rotateAround(rotator, 0, 0, testPressKeyTranslateRot.tempFloatNewRotZ);
//            pPoseStack.rotateAround(rotator, 0, 0, testPressKeyTranslateRot.tempFloatNewZ);
//            pPoseStack.rotateAround(rotator, testPressKeyTranslateRot.tempFloatNewX, testPressKeyTranslateRot.tempFloatNewY, testPressKeyTranslateRot.tempFloatNewZ);
            pPoseStack.rotateAround(rotator, 0, 0, 0);
//            pPoseStack.mulPose(rotator);

            HandEffRotationMatrix.rotateAround(rotator, 0, 0, 0);
        }

        if (bone.yRot != 0) {
            rotator = new Quaternionf();
            Vector3f AxisYP = new Vector3f(0, 1, 0);
            rotator.rotationAxis(bone.yRot, AxisYP);
//            rotator.rotationAxis(MathMethods.AngleToRadiansNoRound(testPressKeyTranslateRot.tempFloatNewRotY), AxisYP);
//            pPoseStack.rotateAround(rotator, 0, testPressKeyTranslateRot.tempFloatNewRotY, 0);

//            pPoseStack.rotateAround(rotator, testPressKeyTranslateRot.tempFloatNewX, testPressKeyTranslateRot.tempFloatNewY, testPressKeyTranslateRot.tempFloatNewZ);
            pPoseStack.rotateAround(rotator, 0, 0, 0);
//            pPoseStack.mulPose(rotator);
            HandEffRotationMatrix.rotateAround(rotator, 0, 0, 0);
        }

        if (bone.xRot != 0) {
            rotator = new Quaternionf();
            Vector3f AxisXN = new Vector3f(1, 0, 0);
            rotator.rotationAxis(bone.xRot, AxisXN);

    }

    private static boolean isShaking(LivingEntity pEntity) {
        return pEntity.isFullyFrozen();
    }

    private static float sleepDirectionToRotation(Direction pFacing) {
        switch (pFacing) {
            case SOUTH -> {
                return 90.0F;
            }
            case WEST -> {
                return 0.0F;
            }
            case NORTH -> {
                return 270.0F;
            }
            case EAST -> {
                return 180.0F;
            }
            default -> {
                return 0.0F;
            }
        }
    }

    private static float getFlipDegrees(LivingEntity pLivingEntity) {
        return 90.0F;
    }

    private void setupRotations(LivingEntity pEntityLiving, PoseStack pMatrixStack, float pRotationYaw, float pPartialTicks, Matrix4f HandEffRotationMatrix) {
        if (isShaking(pEntityLiving)) {
            pRotationYaw += (float)(Math.cos((double)pEntityLiving.tickCount * (double)3.25F) * Math.PI * (double)0.4F);
        }

        if (!pEntityLiving.hasPose(Pose.SLEEPING)) {
            pMatrixStack.mulPose(Axis.YP.rotationDegrees(180.0F - pRotationYaw));
            HandEffRotationMatrix.rotate(Axis.YP.rotationDegrees(180.0F - pRotationYaw));
        }

        if (pEntityLiving.deathTime > 0) {
            float f = ((float)pEntityLiving.deathTime + pPartialTicks - 1.0F) / 20.0F * 1.6F;
            f = MathMethods.sqrt(f);
            if (f > 1.0F) {
                f = 1.0F;
            }

            pMatrixStack.mulPose(Axis.ZP.rotationDegrees(f * getFlipDegrees(pEntityLiving)));
            HandEffRotationMatrix.rotate(Axis.ZP.rotationDegrees(f * getFlipDegrees(pEntityLiving)));
        } else if (pEntityLiving.isAutoSpinAttack()) {
            pMatrixStack.mulPose(Axis.XP.rotationDegrees(-90.0F - pEntityLiving.getXRot()));
            pMatrixStack.mulPose(Axis.YP.rotationDegrees(((float)pEntityLiving.tickCount + pPartialTicks) * -75.0F));
            HandEffRotationMatrix.rotate(Axis.XP.rotationDegrees(-90.0F - pEntityLiving.getXRot()));
            HandEffRotationMatrix.rotate(Axis.YP.rotationDegrees(((float)pEntityLiving.tickCount + pPartialTicks) * -75.0F));
        } else if (pEntityLiving.hasPose(Pose.SLEEPING)) {
            Direction direction = pEntityLiving.getBedOrientation();
            float f1 = direction != null ? sleepDirectionToRotation(direction) : pRotationYaw;
            pMatrixStack.mulPose(Axis.YP.rotationDegrees(f1));
            pMatrixStack.mulPose(Axis.ZP.rotationDegrees(getFlipDegrees(pEntityLiving)));
            pMatrixStack.mulPose(Axis.YP.rotationDegrees(270.0F));
            HandEffRotationMatrix.rotate(Axis.YP.rotationDegrees(f1));
            HandEffRotationMatrix.rotate(Axis.ZP.rotationDegrees(getFlipDegrees(pEntityLiving)));
            HandEffRotationMatrix.rotate(Axis.YP.rotationDegrees(270.0F));
        } else if (LivingEntityRenderer.isEntityUpsideDown(pEntityLiving)) {
            pMatrixStack.translate(0.0F, pEntityLiving.getBbHeight() + 0.1F, 0.0F);
            pMatrixStack.mulPose(Axis.ZP.rotationDegrees(180.0F));
            HandEffRotationMatrix.translate(0.0F, pEntityLiving.getBbHeight() + 0.1F, 0.0F);
            HandEffRotationMatrix.rotate(Axis.ZP.rotationDegrees(180.0F));
        }
    }

    private void scale(LivingEntity pLivingEntity, PoseStack pMatrixStack, float pPartialTickTime, Matrix4f HandEffRotationMatrix) {
        pMatrixStack.scale(0.9375F, 0.9375F, 0.9375F);
        HandEffRotationMatrix.scale(0.9375F, 0.9375F, 0.9375F);
    }
