    private static String getSkillDescription(SkillDataJson skillData, int currentSkillLevel) {
        int correctSkillLevel = 0;
        if (currentSkillLevel > 0) {
            correctSkillLevel = currentSkillLevel - 1;
        }
        double skillDamageMultiplier = getCPlayerSkillDamage();

        int otherSkillLevel = ClientSkillMethods.queryCPlayerCurrentSkillLevel(BerserkerSkillTreeInitialization.BERSERKER_POWER_OF_BLOOD);

        Double TotalHPConsumptionPercentage;
        Double PowerOfBloodMultiplier;

        if (otherSkillLevel > 0) {
            TotalHPConsumptionPercentage = getID2SkillData().get(BerserkerSkillTreeInitialization.BERSERKER_POWER_OF_BLOOD).getValueMultiplierList_01().get(otherSkillLevel - 1);
            PowerOfBloodMultiplier = getID2SkillData().get(BerserkerSkillTreeInitialization.BERSERKER_POWER_OF_BLOOD).getDamageMultiplierList_01().get(otherSkillLevel - 1);
        }
        else {
            TotalHPConsumptionPercentage = 0D;
            PowerOfBloodMultiplier = 0D;
        }

        Double DamageMultiplier = skillData.getDamageMultiplierList_01().get(correctSkillLevel);
        Double ConstantDamage = skillData.getDamageList_01().get(correctSkillLevel);

        Float finalDmg = (float) ( ( (DamageMultiplier + BerserkerSkillTreeInitialization.BERSERKER_TOTAL_HP_CONSUMPTION_MULTIPLIER_2 * TotalHPConsumptionPercentage + PowerOfBloodMultiplier) * getEntityAttackDamage(getClientPlayer()) + ConstantDamage) * skillDamageMultiplier);

        Double GroundWaveDamageMultiplier = skillData.getDamageMultiplierList_02().get(correctSkillLevel);
        Double GroundWaveConstantDamage = skillData.getDamageList_02().get(correctSkillLevel);

        Float finalAdditionalDmg = (float) ( ( (GroundWaveDamageMultiplier + BerserkerSkillTreeInitialization.BERSERKER_TOTAL_HP_CONSUMPTION_MULTIPLIER_2 * TotalHPConsumptionPercentage + PowerOfBloodMultiplier) * getEntityAttackDamage(getClientPlayer()) + GroundWaveConstantDamage) * skillDamageMultiplier);

        Double GroundAOE = skillData.getEffectArea().get(correctSkillLevel);

        return LocaleLanguageMethods.getTranslatableDescriptionWithArgs(skillData.getTranslatableTextDescription(),
                String.format("%d%%", (int) (TotalHPConsumptionPercentage * 100)),
                String.format("%d%%", (int) (DamageMultiplier.floatValue() * 100)),
                String.format("%s", ConstantDamage),
                String.format("%s", finalDmg),
                String.format("%d%%", (int) (GroundWaveDamageMultiplier.floatValue() * 100)),
                String.format("%s", GroundWaveConstantDamage),
                String.format("%s", finalAdditionalDmg),
                String.format("%s", GroundAOE)

        );
    }

    public static boolean Berserker013_SkillClientCondition(String SkillID) {
        boolean isInCD = isCPlayerSkillInCD(SkillID);
        boolean isInAction = isPlayerInAction();
        boolean isChanneling = isPlayerInChannelingState();
        boolean isOnGround = isPlayerOnGround();
        boolean isPlayerBeingHit = isPlayerBeingHit();

        boolean isTriggeredLastAttack = ClientPlayerMethods.getClientPlayerStateAmount(Berserker013_IsLastAttackAvailable) == Constants.ACTION_ON;

        return isTriggeredLastAttack || (!isInCD && !isInAction && !isChanneling && isOnGround && !isPlayerBeingHit);
    }

    public static void init() {
        SkillDataJson tempSkillData = getID2SkillData().get(SkillID);
        tempSkillData.setTranslatableTooltipGetter(Berserker013::getSkillDescription);

        //Condition
        tempSkillData.setClientConditionRequirement(Berserker013::Berserker013_SkillClientCondition);
//        tempSkillData.setClientCondReqPassedAction(DfoSwordmanSkillMethods::defaultSkillClientConditionPassedAction);

        if (isClientSide()) {
            return;
        }

        //====================================
        //setup skill action
        MethodAction tempSkillAction =
                (entity1) -> {
                    if (!SBSTickEventMethods.hasSkillActionRunning(entity1, Berserker013_PrepAction)) {
                        if (Berserker013_BloodSwordEntity.get(entity1) != null) {
                            printInGameMsg("start");
                            SBSTickEventMethods.setSkillActionDone(entity1, SkillID);
                            Berserker013_BloodSwordEntity.remove(entity1);
                            Berserker013_TriggerFinalAction(entity1);
                            setEntityStateAmount(entity1, Berserker013_IsLastAttackAvailable, Constants.ACTION_OFF);
                        } else {
                            int playerSkillLVL = 0;

                            SkillDataJson skill1 = getID2SkillData().get(SkillID);

                            if (entity1 instanceof ServerPlayer sp1) {
                                SBPlayerAnimatorMethods.PlayDfoSwdSkillAnimToAllClientPlayers(sp1, BERSERKER013_SLASH_DOWNWARD_PREPARATION_IN_MID_ANIMATION);
                                playerSkillLVL = ServerSkillMethods.querySPlayerCurrentSkillLevel(sp1, SkillID);
                                if (playerSkillLVL <= 0) {
                                    return;
                                }
                                playerSkillLVL = playerSkillLVL - 1;
                            }

                            float triggeredEntityHorizontalFacingAngle = getEntityHorizontalFacingDeg(entity1);

                            incrementEntityStateByAmountX(entity1, Constants.IS_IN_ACTION, 1);

                            int expireTicks = (int) (skill1.getValueUsage_01().get(playerSkillLVL).floatValue() * 20);
                            int TotalChannelingTicks = (int) (skill1.getChannelingTime().get(playerSkillLVL).floatValue() * 20);
                            int actionPeriod = 2;
                            IntegerValueHolder ElapsedTicksHolder = new IntegerValueHolder(0);

                            GeoModelSpawnMethods.Generate2DAnimatedTexCustomEntity256BoundToEntity(entity1, 255, 255, 255, 1f, 0, 1.4f, 0, 900, GeoAnimationAndTextureName.EFF_COMMON_CHARGING_LOOPING, DfoSwordmanSkillTreeConstants.NO_ANIMATION,
                                    DfoSwordmanSkillTreeConstants.NO_ANIMATION, 0, 0, 13, false, false, false, DfoSwordmanSkillMethods.RemoveEffEntityIfNotInAction(entity1), 5, true);

                            //Berserker013_PrepAction
                            MethodAction action = (entity2) -> {
                                int ElapsedTicks = ElapsedTicksHolder.getIntCounter();
                                ElapsedTicksHolder.setIntCounter(ElapsedTicks + actionPeriod);

                                if (ElapsedTicks >= TotalChannelingTicks) {
                                    SBSTickEventMethods.setSkillActionDone(entity2, Berserker013_PrepAction);
                                    incrementEntityStateByAmountXNoPacket(entity2, Constants.IS_IN_ACTION, -1);
                                    setEntityStateAmount(entity2, Berserker013_IsLastAttackAvailable, Constants.ACTION_ON);

                                    ExtremeOverkillBloodSwordEntity bloodSwordEntity = BerserkerModelSpawnMethods.GenerateExtremeOverkillBloodSword(entity2, triggeredEntityHorizontalFacingAngle, 1f, expireTicks, 0, 0, 0,
                                            Berserker013_TriggerBloodSwordAction(ElapsedTicksHolder, expireTicks), actionPeriod, true);
                                    Berserker013_BloodSwordEntity.put(entity2, bloodSwordEntity);

                                    IntegerValueHolder ElapsedTicksHolder2 = new IntegerValueHolder(0);
                                    MethodAction action2 = (entity3) -> {
                                        int ElapsedTicks2 = ElapsedTicksHolder2.getIntCounter();
                                        ElapsedTicksHolder2.setIntCounter(ElapsedTicks2 + actionPeriod);
                                        //expired
                                        if (ElapsedTicks2 >= expireTicks) {
                                            printInGameMsg("expired");
                                            SBSTickEventMethods.setSkillActionDone(entity3, SkillID);
                                            Berserker013_BloodSwordEntity.remove(entity3);
                                            Berserker013_TriggerFinalAction(entity3);
                                            setEntityStateAmount(entity3, Berserker013_IsLastAttackAvailable, Constants.ACTION_OFF);
                                        }
                                    };

                                    SBSTickEventMethods.setMethodActionTimer(entity2, SkillID, actionPeriod, action2);

                                } else {
                                    if (isLivingEntityActionInterrupted(entity2)) {
                                        incrementEntityStateByAmountXNoPacket(entity2, Constants.IS_IN_ACTION, -1);
                                        SBSTickEventMethods.setSkillActionDone(entity2, Berserker013_PrepAction);
                                    }
                                }
                            };


                            SBSTickEventMethods.setMethodActionTimer(entity1, Berserker013_PrepAction, actionPeriod, action);
                        }
                    }
                };

        tempSkillData.setActiveSkillAction1(tempSkillAction);
    }

    public static MethodEntityAction Berserker013_TriggerBloodSwordAction(IntegerValueHolder ElapsedTicksHolder, int expireTicks) {
        return (tempBloodSwordEntity) -> {
            ExtremeOverkillBloodSwordEntity bloodSwordEntity = (ExtremeOverkillBloodSwordEntity) tempBloodSwordEntity;
            Entity owner = bloodSwordEntity.getOwner();
            if (owner instanceof LivingEntity livingEntity) {
                boolean isTriggeredLastAttack = ServerSkillMethods.getEntityStateAmount(livingEntity, Berserker013_IsLastAttackAvailable) == Constants.ACTION_OFF;
                if (isTriggeredLastAttack) {
                    RemoveEntity(bloodSwordEntity);
                }
            }
            if (ElapsedTicksHolder.getIntCounter() >= expireTicks) {
                RemoveEntity(bloodSwordEntity);
            }

        };
    }

    public static void Berserker013_TriggerFinalAction(LivingEntity triggeredEntity) {
        //ground damage action
        String GroupDmgActionID = BERSERKER013_GROUND_DAMAGE_ACTION + getEntityTickCount(triggeredEntity);
        MethodAction groundDamageAction = Berserker013_DealerBloodSwordExplosionAction(triggeredEntity, BERSERKER013_GROUND_DAMAGE_ACTION_PERIOD, GroupDmgActionID);
        SBSTickEventMethods.setMethodActionTimer(triggeredEntity, GroupDmgActionID, BERSERKER013_GROUND_DAMAGE_ACTION_PERIOD, groundDamageAction);
    }

    public static MethodAction Berserker013_DealerBloodSwordExplosionAction(LivingEntity triggeredEntity, int actionPeriod, String ActionID) {
        int playerSkillLVL = 0;

        float finalAOE;
        float damageBase;
        float damageMultiplier;
        float skillDamageMultiplier = 1f + SBSAttributeMethods.getSkillDamageAttrVal(triggeredEntity);
        Predicate<Entity> finalCondition;

        SkillDataJson skill1 = getID2SkillData().get(SkillID);

        float ExtraDamageMultiplier = 0;

        if (triggeredEntity instanceof ServerPlayer sp1) {
            finalCondition = defaultSkillTargetCondition(sp1).and(SBConditions.isNotTargetEntity(triggeredEntity));
            SBPlayerAnimatorMethods.PlayDfoSwdSkillAnimToAllClientPlayers(sp1, BERSERKER013_SLASH_DOWNWARD_PREPARATION_IN_MID_ANIMATION);
            playerSkillLVL = ServerSkillMethods.querySPlayerCurrentSkillLevel(sp1, SkillID);
            playerSkillLVL = playerSkillLVL - 1;
        } else {
            finalCondition = defaultSkillTargetCondition().and(SBConditions.isNotTargetEntity(triggeredEntity));
        }
        int DurationTicks = (int) (skill1.getDuration().get(playerSkillLVL).floatValue() * 20);

        finalAOE = skill1.getEffectArea().get(playerSkillLVL).floatValue();

        damageBase = skill1.getDamageList_01().get(playerSkillLVL).floatValue();
        damageMultiplier = skill1.getDamageMultiplierList_01().get(playerSkillLVL).floatValue() + ExtraDamageMultiplier;

        float finalDmg = (float) ((damageMultiplier * getEntityAttackDamage(triggeredEntity) + damageBase) * skillDamageMultiplier);

        float GroundDamageBase = skill1.getDamageList_02().get(playerSkillLVL).floatValue();
        float GroundDamageMultiplier = skill1.getDamageMultiplierList_02().get(playerSkillLVL).floatValue() + ExtraDamageMultiplier;

        float finalPillarDmg = (float) ((GroundDamageMultiplier * getEntityAttackDamage(triggeredEntity) + GroundDamageBase) * skillDamageMultiplier);

        IntegerValueHolder ElapsedTicksHolder = new IntegerValueHolder(0);
        Vector3f centerPos = getEntityPos(triggeredEntity);
        float triggeredEntityHorizontalFacingDeg = getEntityHorizontalFacingDeg(triggeredEntity);
        int damagePeriod = 4;

        float effEntityScale = DfoSwordmanSkillMethods.getEntityScaleByAOE(finalAOE);

        GeoModelSpawnMethods.Generate2DAnimatedTexCustomFlatEntity256(triggeredEntity, triggeredEntityHorizontalFacingDeg, 255, 255, 255, finalAOE, 0, 0.25f, 0, 6, false,
                centerPos.x(), centerPos.y(), centerPos.z(), centerPos.x(), centerPos.y(), centerPos.z(), Berserker013_EXTREME_OVERKILL_SHOCK_WAVE_TEXTURE_PATH, DfoSwordmanSkillTreeConstants.NO_ANIMATION, DfoSwordmanSkillTreeConstants.NO_ANIMATION, DfoSwordmanSkillTreeConstants.NO_ANIMATION,
                1, false, 0, 0, 3,
                false, false, true, null, 777, true);

        GeoModelSpawnMethods.Generate2DAnimatedTexCustomFlatEntity256(triggeredEntity, triggeredEntityHorizontalFacingDeg, 255, 255, 255, finalAOE, 0, 0.1f, 0, BERSERKER013_GROUND_DAMAGE_ACTION_PREPARATION_TICKS + 2, false,
                centerPos.x(), centerPos.y(), centerPos.z(), centerPos.x(), centerPos.y(), centerPos.z(), Berserker013_EXTREME_OVERKILL_GROUND_CRACK_TEXTURE_PATH, DfoSwordmanSkillTreeConstants.NO_ANIMATION, DfoSwordmanSkillTreeConstants.NO_ANIMATION, DfoSwordmanSkillTreeConstants.NO_ANIMATION,
                1, false, 0, 0, 5,
                false, false, true, null, 777, true);

        BooleanValueHolder effEntitySpawnedYet = new BooleanValueHolder(false);
        TypeValueHolder<Entity> effEntityHolder = new TypeValueHolder<>(null);

        return (dealer) -> {
            int ElapsedTicks = ElapsedTicksHolder.getIntCounter();
            ElapsedTicksHolder.setIntCounter(ElapsedTicks + actionPeriod);

            if (ElapsedTicks >= BERSERKER013_GROUND_DAMAGE_ACTION_PREPARATION_TICKS) {
                if (!effEntitySpawnedYet.getBooleanVal()) {
                    effEntitySpawnedYet.setBooleanVal(true);
                    BerserkerModelSpawnMethods.GenerateExtremeOverkillBackGround(triggeredEntity, triggeredEntityHorizontalFacingDeg, effEntityScale, DurationTicks, 0, 0, 0, null, 777, true);

                    GeoModelSpawnMethods.Generate2DAnimatedTexCustomFlatEntity256(triggeredEntity, triggeredEntityHorizontalFacingDeg, 255, 255, 255, finalAOE, 0, 0.1f, 0, DurationTicks, false,
                            centerPos.x(), centerPos.y(), centerPos.z(), centerPos.x(), centerPos.y(), centerPos.z(), Berserker013_EXTREME_OVERKILL_GROUND_BACKGROUND_TEXTURE_PATH, DfoSwordmanSkillTreeConstants.NO_ANIMATION, DfoSwordmanSkillTreeConstants.NO_ANIMATION, DfoSwordmanSkillTreeConstants.NO_ANIMATION,
                            1, false, 0, 0, 20,
                            false, false, false, null, 777, true);

                    GeoModelSpawnMethods.Generate2DAnimatedTexCustomFlatEntity256(triggeredEntity, triggeredEntityHorizontalFacingDeg, 255, 255, 255, finalAOE, 0, BERSERKER013_GROUND_DAMAGE_ACTION_BLOOD_PILLAR_HEIGHT - 0.1f, 0, DurationTicks, false,
                            centerPos.x(), centerPos.y(), centerPos.z(), centerPos.x(), centerPos.y(), centerPos.z(), Berserker013_EXTREME_OVERKILL_GROUND_BACKGROUND_TEXTURE_PATH, DfoSwordmanSkillTreeConstants.NO_ANIMATION, DfoSwordmanSkillTreeConstants.NO_ANIMATION, DfoSwordmanSkillTreeConstants.NO_ANIMATION,
                            1, false, 0, 0, 20,
                            false, false, false, null, 777, true);
                    Entity effEntity = DfoGeneralEntitySpawnMethods.SetupClientParticleHelperNoLoc(triggeredEntity, 0, finalAOE, DurationTicks, centerPos.x(), centerPos.y(), centerPos.z(), DfoSwordmanSkillTreeConstants.NO_ANIMATION, Berserker013_PILLAR_PARTICLE, null, 777, true);
                    effEntityHolder.setAnyTypeValue(effEntity);

                }
                if (ElapsedTicks <= DurationTicks + BERSERKER013_GROUND_DAMAGE_ACTION_PREPARATION_TICKS) {
                    if (ElapsedTicks % damagePeriod == 0) {
                        Entity effEntity = effEntityHolder.getAnyTypeValue();
                        if (effEntity != null) {
                            //start damage
                            float entityPosX, entityPosZ, entityTowardsTargetLocAngle, newDX, newDZ;
                            for (Entity tempEntity : GroupEnumEntitiesInRangeFilter(effEntity, finalAOE, 1, finalAOE, finalAOE, BERSERKER013_GROUND_DAMAGE_ACTION_BLOOD_PILLAR_HEIGHT, finalAOE, finalCondition)) {
                                if (tempEntity instanceof LivingEntity tempLivingEntity) {
                                    DfoSwdSoundMethods.PlayBooDongHitSound(getEntityLevel(dealer), getEntityX(tempEntity), getEntityY(tempEntity), getEntityZ(tempEntity), 0.3f, 1.0f);

                                    entityPosX = getEntityX(tempLivingEntity);
                                    entityPosZ = getEntityZ(tempLivingEntity);

                                    entityTowardsTargetLocAngle = Vector3fMethods.AngleFromLocationToLocationHorizontalInRoundedDegree(entityPosX, entityPosZ, centerPos.x(), centerPos.z());
                                    newDX = Vector3fMethods.PolarProjectionFromXHorizon(0.25f, entityTowardsTargetLocAngle);
                                    newDZ = Vector3fMethods.PolarProjectionFromZHorizon(0.25f, entityTowardsTargetLocAngle);
//                                        float pushYStrength = EntityMethods.calculatePushEntityStrength(0.18f, dealer, tempLivingEntity);
//                                    EntityMethods.pushEntityWithDeltaValue(tempLivingEntity, 0, 0, pushYStrength, false);
                                    setEntitySPDeltaMovement(tempLivingEntity, newDX, 0.25f, newDZ, false);

                                    DfoGeneralDamageSrc.LivingEntityDamageTargetWithPhysicDMG(dealer, tempLivingEntity, finalPillarDmg, false);

                                    DfoStatusGeneralMethods.ApplyBleedingStatusToTargetLivingEntity(dealer, tempLivingEntity, DfoStatusGeneralMethods.BLEEDING_TOTAL_DURATION_TICKS, finalDmg * DfoStatusGeneralMethods.BLEEDING_DEFAULT_DAMAGE_MULTIPLIER);
                                }
                            }
                        }
                        else {
                            SBSTickEventMethods.setSkillActionDone(dealer, ActionID);
                        }
                    }
                } else {
                    SBSTickEventMethods.setSkillActionDone(dealer, ActionID);
                }
            }
        };
    }

    public static void Berserker013_resetEntityState(LivingEntity entity) {
        setEntityStateAmount(entity, Berserker013_IsLastAttackAvailable, Constants.ACTION_OFF);
        Entity swordEntity = Berserker013_BloodSwordEntity.remove(entity);
        if (swordEntity != null) {
            RemoveEntity(swordEntity);
        }
    }

    public static void Berserker013_collectBloodOrbFromEntities(LivingEntity beingAttacked, LivingEntity attacker) {
        ExtremeOverkillBloodSwordEntity attackerBloodSwordEntity = Berserker013_BloodSwordEntity.get(attacker);

        if (attackerBloodSwordEntity != null) {
            Vector3f beingAttackedEntityPos = getEntityPos(beingAttacked);
            int randomHorizontalFacingInt = ExtraMathMethods.randomInt(360);

            Entity effEntity = GeoModelSpawnMethods.Generate2DAnimatedTexCustomEntity256(attackerBloodSwordEntity,255, randomHorizontalFacingInt + 360, randomVerticalFacingInt + 360, 0.5f, 0, 0f, 0, 200, false,

    public static MethodEntityAction Berserker013_OrbAbsorptionServerAction(FloatValueHolder distHasTraveled, FloatValueHolder horizontalFacing, FloatValueHolder verticalFacing, FloatValueHolder initialAcceleration, TypeValueHolder<Vector3f> previousTargetLoc) {
        return (tempOrbEntity) -> {
            TwoDAnimated256TexCustomEntity OrbEntity = (TwoDAnimated256TexCustomEntity) tempOrbEntity;
            int currentSwordEntityElapsedLifeTime = OrbEntity.getMaxLifetime() - OrbEntity.getLifetime();
            ExtremeOverkillBloodSwordEntity owner = (ExtremeOverkillBloodSwordEntity) OrbEntity.getOwner();
            if (owner != null && !SBConditions.isEntityAlive(owner)) {
                owner = null;
            }
            if (owner != null) {
                Vector3f originEntityLoc = (previousTargetLoc.getAnyTypeValue() == null) ? getEntityPos(OrbEntity) : previousTargetLoc.getAnyTypeValue();

                float distShouldTravelEachTime = initialAcceleration.getFloatVal();
                float polarDist = presetDist1 - distShouldTravelEachTime;
                float orbHorizontalFacingAngle;
                float orbVerticalFacingAngle;
                boolean reachedMaxDist = false;

                if (distHasTraveled.getFloatVal() < Berserker013_BloodOrbMaxDist) {
                    distHasTraveled.setFloatVal(distHasTraveled.getFloatVal() + polarDist);
                    orbHorizontalFacingAngle = horizontalFacing.getFloatVal();
                    orbVerticalFacingAngle = verticalFacing.getFloatVal();
                }
                else {
                    reachedMaxDist = true;
                    Vector3f targetEntityLoc = getEntityPos(owner);
                    orbHorizontalFacingAngle = Vector3fMethods.AngleFromLocationToLocationHorizontalInRoundedDegree(originEntityLoc, targetEntityLoc);
                    orbVerticalFacingAngle = Vector3fMethods.convertNormalizedVerticalAngleToMCVerticalFacingAngle(MathMethods.RoundAngle(Vector3fMethods.AngleFromLocationToLocationVerticalInRad(originEntityLoc, targetEntityLoc) * MathMethods.RAD_TO_DEG));
                }
                initialAcceleration.setFloatVal(distShouldTravelEachTime - distShouldTravelDecay);

                Vector3f tpLoc = Vector3fMethods.PolarProjectionFromRotation(originEntityLoc, MathMethods.min(0.25f, polarDist), orbHorizontalFacingAngle, orbVerticalFacingAngle);

                moveEntityToPos(OrbEntity, tpLoc);

                previousTargetLoc.setAnyTypeValue(tpLoc);

//                Vector3f targetLoc = EntityMethods.getEntityBlockPos(owner).add(0, 1.4f, 0);
                Vector3f targetLoc = getEntityPos(owner).add(0, 1.4f, 0);

                if (Vector3fMethods.Vec3fDistanceTo(tpLoc.x(), tpLoc.y() + 1.4f, tpLoc.z(), targetLoc.x(), targetLoc.y(), targetLoc.z()) < 0.5f) {
                    //TODO test
                    RemoveEntity(OrbEntity);
                    owner.setExtremeOverkillServerAbsorptionCount(owner.getExtremeOverkillServerAbsorptionCount() + 1);
//                    if (owner instanceof LivingEntity livingEntity) {
//
//                    }

                    GeoModelSpawnMethods.Generate2DAnimatedTexCustomEntity256BoundToEntity(owner, 255, 255, 255, 1f, 0, 1.4f, 0, 12, GeoAnimationAndTextureName.EFF_RECOVERY_TEXTURE_PATH, DfoSwordmanSkillTreeConstants.NO_ANIMATION,
                            DfoSwordmanSkillTreeConstants.NO_ANIMATION,0, 0, 12, false, false, true, null, 777, true);
                }
                else if (!reachedMaxDist) {
                    float HorizontalAngleBetweenSwordEntityAndTarget = Vector3fMethods.AngleFromLocationToLocationHorizontalInRoundedDegree(originEntityLoc, targetLoc);
                    float VerticalAngleBetweenSwordEntityAngTarget = MathMethods.RoundAngle(Vector3fMethods.AngleFromLocationToLocationVerticalInDegrees(originEntityLoc, targetLoc));

                    float time = 0.01f + 0.99f * currentSwordEntityElapsedLifeTime / OrbEntity.getMaxLifetime();
                    float newHorizontalDegree = MathMethods.RoundAngle(MathMethods.lerpDegreeAngle(time, orbHorizontalFacingAngle, HorizontalAngleBetweenSwordEntityAndTarget));
                    float newVerticalDegree = MathMethods.RoundAngle(MathMethods.lerpDegreeAngle(time, MathMethods.RoundAngle(orbVerticalFacingAngle), VerticalAngleBetweenSwordEntityAngTarget));
                    horizontalFacing.setFloatVal(newHorizontalDegree);
                    verticalFacing.setFloatVal(newVerticalDegree);
                }
            }
        };
    }

    public static final int ParticleLifeTime = 50;

    public static void Berserker013_ParticleRegistry(HashMap<String, MethodEntityAction> particleMap) {

    public static void CreateBloodPillarParticleMethodReference(Entity entity) {
        ClientParticleHelperEntity clientParticleHelperEntity = (ClientParticleHelperEntity) entity;
        Vector3f center = getEntityPos(clientParticleHelperEntity);
        float aoeRange = getEntityVerticalFacingDeg(clientParticleHelperEntity);

        int minX = (int) (center.x() - aoeRange);
        int minZ = (int) (center.z() - aoeRange);
        int maxX = (int) (center.x() + aoeRange);
        int maxZ = (int) (center.z() + aoeRange);
        float centerY = center.y() + 0.1f;
        float centerY2 = center.y() - 0.9f;
//        Vector3f randomDelta;

        float x1, z1, dist, xi, zi;
        for (int i = minX + 1; i < maxX; i++) {
            for (int j = minZ + 1; j < maxZ; j++) {
                xi = i + 0.5f;
                zi = j + 0.5f;
                x1 = Math.abs(xi - center.x());
                z1 = Math.abs(zi - center.z());
                dist = MathMethods.sqrt(x1 * x1 + z1 * z1);

                if (dist <= aoeRange) {
                    ParticleMethods.createParticleByTypeAtLoc(ParticleTypes.WHITE_ASH, xi, centerY, zi,
                            0, ParticleSpeed, 0,
                            0.33f, 3, ParticleLifeTime, 0.4f, 0, 0);

                    ParticleMethods.createParticleByTypeAtLoc(ParticleTypes.WHITE_ASH, xi, centerY2, zi,
                            0, ParticleSpeed, 0,
                            0.33f, 3, ParticleLifeTime, 0.4f, 0, 0);
                }
            }
        }
    }
